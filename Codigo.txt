class Block {
  constructor(index, data, previousHash = '') {
    this.index = index;
    this.timestamp = new Date();
    this.data = data;
    this.previousHash = previousHash;
    this.hash = this.calculateHash();
  }

  calculateHash() {
    return this.index + this.previousHash + this.timestamp + JSON.stringify(this.data);
  }
}

class Blockchain {
  constructor() {
    this.chain = [this.createGenesisBlock()];
  }

  createGenesisBlock() {
    return new Block(0, "Bloque Génesis", "0");
  }

  getLatestBlock() {
    return this.chain[this.chain.length - 1];
  }

  addBlock(newBlock) {
    newBlock.previousHash = this.getLatestBlock().hash;
    newBlock.hash = newBlock.calculateHash();
    this.chain.push(newBlock);
  }
}

// Simulación
let blockchain = new Blockchain();

blockchain.addBlock(new Block(1, { mensaje: "Hola desde Servidor A" }));
blockchain.addBlock(new Block(2, { mensaje: "Respuesta desde Servidor B" }));

console.log("BLOCKCHAIN GENERADA:");
console.log(blockchain);
